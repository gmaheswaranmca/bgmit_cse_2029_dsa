class Solution {
    public int minEatingSpeed(int[] piles, int h) {

        int left = 1;
        int right = 0;

        // Find maximum pile (maximum possible eating speed)
        for (int pile : piles) {
            right = Math.max(right, pile);
        }

        // Binary Search to find minimum valid eating speed
        while (left < right) {

            // Find middle eating speed
            int mid = left + (right - left) / 2;

            // Calculate total hours required at speed mid
            long hours = 0;

            for (int pile : piles) {
                // Ceiling division: hours needed for each pile
                hours += (pile + mid - 1) / mid;
            }

            // If Koko can finish within h hours,
            // try a smaller eating speed
            if (hours <= h) {
                right = mid;
            }
            // Otherwise, increase the eating speed
            else {
                left = mid + 1;
            }
        }

        // Minimum valid eating speed
        return left;
    }
}
